
1. Swapping Two Numbers Without Using a Third Variable Procedure Creation

CREATE OR REPLACE PROCEDURE swap_number (
    a IN OUT NUMBER,
    b IN OUT NUMBER
) AS
BEGIN
    a := a + b;
    b := a - b;
    a := a - b;

    DBMS_OUTPUT.PUT_LINE('After Swapping:');
    DBMS_OUTPUT.PUT_LINE('A = ' || a);
    DBMS_OUTPUT.PUT_LINE('B = ' || b);
END;


SET SERVEROUTPUT ON

DECLARE
    x NUMBER := 10;
    y NUMBER := 20;
BEGIN
    DBMS_OUTPUT.PUT_LINE('Before Swapping:');
    DBMS_OUTPUT.PUT_LINE('X = ' || x);
    DBMS_OUTPUT.PUT_LINE('Y = ' || y);

    swap_number(x, y);
END;
/

Finding the Largest Amongst Three Numbers Procedure Creation: CREATE OR REPLACE PROCEDURE FindLargest

CREATE OR REPLACE PROCEDURE FindLargest (
    a IN NUMBER,
    b IN NUMBER,
    c IN NUMBER
) AS
    largest NUMBER;
BEGIN
    IF a >= b AND a >= c THEN
        largest := a;
    ELSIF b >= a AND b >= c THEN
        largest := b;
    ELSE
        largest := c;
    END IF;

    DBMS_OUTPUT.PUT_LINE('Largest number is: ' || largest);
END;
/

SET SERVEROUTPUT ON

BEGIN
    FindLargest(10, 25, 15);
END;
/



SET SERVEROUTPUT ON

BEGIN
    FindLargest(10, 25, 15);
END;
/

Calculating the Area of a Circle Given the Radius Procedure Creation

CREATE OR REPLACE PROCEDURE Area (
    r IN NUMBER,
    area OUT NUMBER
) AS
BEGIN
    area:= 3.14159 * r * r;
END;
/


DECLARE
    radius NUMBER := 7;
    area NUMBER;
BEGIN
    Area(radius, area);
    DBMS_OUTPUT.PUT_LINE('Area of the Circle = ' || area);
END;
/
